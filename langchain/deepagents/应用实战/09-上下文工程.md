> 🧭 **双语说明**：标 🇹🇸 的段落是 TS 侧实测补充。两侧的**机制**（四段流、三档压缩、卸载阈值）几乎逐条对齐，差异集中在三个地方：① 压缩后的条数计数（TS 不把 system 算进 messages）；② 卸载只认 `ToolMessage`；③ 缓存中间件按 provider 装配，名单不同。

## 场景：给长任务配一套"上下文预算"

**场景**：你的 agent 要跑一个调研任务，会读几十个文件、调几个外部 API、写几份报告。你担心两件事：

1. **崩**：上下文撑爆，任务半路死掉
2. **变傻变贵**：没崩，但越跑越慢、越来越贵、越来越不听话

**全貌一句话**：完整方案还包括分布式追踪（看每次压缩何时发生）、按任务类型设不同阈值、以及把压缩事件接入告警。本课解决"看懂它怎么瘦、自己会调旋钮"这一段。

### ① 基础实现（能跑但不知道它怎么瘦的）

![基础版：制造三种膨胀，看三种压缩](./assets/app-step1-context-inflate.svg)

> 看图：中间那条是上下文占用曲线，三个箭头分别是三种膨胀源（大工具结果、大写入参数、超长对话）。每个箭头指向救它的机制。注意右上角那个红框——`read_file` 的箭头被划掉了，它不靠卸载救命。

```python [Python]
import tempfile
from pathlib import Path

from langchain_core.language_models.chat_models import BaseChatModel
from langchain_core.messages import AIMessage
from langchain_core.outputs import ChatGeneration, ChatResult
from langgraph.checkpoint.memory import InMemorySaver

from deepagents import create_deep_agent
from deepagents.backends import FilesystemBackend


class Spy(BaseChatModel):
    """记录每次送给模型的消息，不发网络请求。"""

    @property
    def _llm_type(self):
        return "spy"

    def bind_tools(self, tools, **kwargs):
        return self

    def _generate(self, messages, stop=None, run_manager=None, **kwargs):
        SENT.append(list(messages))
        return ChatResult(generations=[ChatGeneration(message=AIMessage(content="ok"))])


SENT = []

root = Path(tempfile.mkdtemp())
fsb = FilesystemBackend(root_dir=str(root), virtual_mode=True)
agent = create_deep_agent(model=Spy(), backend=fsb, checkpointer=InMemorySaver())
```

```typescript [TypeScript]
import { BaseChatModel } from "@langchain/core/language_models/chat_models";
import type { BaseMessage } from "@langchain/core/messages";
import { AIMessage } from "@langchain/core/messages";
import type { ChatResult } from "@langchain/core/outputs";
import { MemorySaver } from "@langchain/langgraph";
import { mkdtempSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { createDeepAgent, FilesystemBackend } from "deepagents";

const SENT: BaseMessage[][] = [];

class Spy extends BaseChatModel {
  constructor() { super({}); }
  _llmType() { return "spy"; }
  bindTools() { return this; }
  async _generate(messages: BaseMessage[]): Promise<ChatResult> {
    SENT.push([...messages]);
    return { generations: [{ text: "ok", message: new AIMessage("ok") }] } as ChatResult;
  }
}

const root = mkdtempSync(join(tmpdir(), "ctx-"));
const fsb = new FilesystemBackend({ rootDir: root, virtualMode: true });
const agent = await createDeepAgent({
  model: new Spy() as never,
  backend: fsb,
  checkpointer: new MemorySaver(),
});
```

:::

实测：无 memory、无 skills 时，系统提示约 **2,600 字符**（内置 agent 指南）。加上 memory 会多约 5,118，加 skills 会多约 1,900。

> 🇹🇸 TS 侧的数字同量级（memory 固定税 4,470 字符），差别只在提示模板的措辞长度。**量法完全一样**——用假模型抓 `messages[0]` 就够，不需要靠日志埋点。

> ⚠️ **它的三个问题**：
> 1. **不知道什么时候会压缩**。默认阈值随模型变（200k 窗口 → 170k 触发），你从来没量过。
> 2. **不知道压缩有没有发生**。跑完了，但中间瘦过几次？不知道。
> 3. **不知道哪个工具会被卸载**。以为所有大结果都安全，其实文件工具根本不在名单里。

**变化一：强制触发压缩，看清它做了什么**

::: code-group

```python [Python]
from deepagents.middleware.summarization import SummarizationMiddleware
from langchain_core.messages import HumanMessage


class SummaryModel(BaseChatModel):
    """摘要用的小模型。"""

    @property
    def _llm_type(self):
        return "summarizer"

    def _generate(self, messages, stop=None, run_manager=None, **kwargs):
        return ChatResult(generations=[ChatGeneration(
            message=AIMessage(content="【摘要】用户让我读了若干文件，结论是 A。")
        )])


summ = SummarizationMiddleware(
    model=SummaryModel(),
    backend=fsb,
    trigger=("tokens", 200),      # 故意极小，方便观察
    keep=("messages", 2),
)

agent2 = create_deep_agent(
    model=Spy(), backend=fsb, middleware=[summ], checkpointer=InMemorySaver()
)

msgs = []
for i in range(8):
    msgs.append(HumanMessage(content=f"第 {i} 轮：读 src/module_{i}.py。" + "细节" * 40))
    msgs.append(AIMessage(content=f"第 {i} 轮回答：module_{i} 处理数据流。" + "分析" * 40))

SENT.clear()
out = agent2.invoke({"messages": msgs}, config={"configurable": {"thread_id": "T1"}})

print(f"输入 {len(msgs)} 条 → 送给模型 {len(SENT[0])} 条")
print(f"state 里仍是 {len(out['messages'])} 条（原文保留）")
print(f"存档文件：{list((root / 'conversation_history').iterdir())}")
```

```typescript [TypeScript]
import { createSummarizationMiddleware } from "deepagents";
import { HumanMessage } from "@langchain/core/messages";

class SummaryModel extends BaseChatModel {
  constructor() { super({}); }
  _llmType() { return "summarizer"; }
  async _generate(): Promise<ChatResult> {
    return { generations: [{ text: "s", message: new AIMessage("【摘要】用户让我读了若干文件，结论是 A。") }] } as ChatResult;
  }
}

const summ = createSummarizationMiddleware({
  model: new SummaryModel() as never,
  backend: fsb,
  trigger: { type: "tokens", value: 200 },   // 故意极小，方便观察
  keep: { type: "messages", value: 2 },
});

const agent2 = await createDeepAgent({
  model: new Spy() as never,
  backend: fsb,
  middleware: [summ],
  checkpointer: new MemorySaver(),
});

const msgs: BaseMessage[] = [];
for (let i = 0; i < 8; i++) {
  msgs.push(new HumanMessage(`第 ${i} 轮：读 src/module_${i}.py。` + "细节".repeat(40)));
  msgs.push(new AIMessage(`第 ${i} 轮回答：module_${i} 处理数据流。` + "分析".repeat(40)));
}

SENT.length = 0;
const out = await agent2.invoke({ messages: msgs }, { configurable: { thread_id: "T1" } });

// 注意：TS 的 SENT 会收到 2 次调用——第 0 次是"给摘要模型"，最后一次才是主调用
const main = SENT[SENT.length - 1];
console.log(`输入 ${msgs.length} 条 → 主调用送给模型 ${main.length} 条（共 ${SENT.length} 次模型调用）`);
console.log(`state 里仍是 ${(out.messages as BaseMessage[]).length} 条（原文保留）`);
console.log(`存档文件：${readdirSync(join(root, "conversation_history")).join(", ")}`);
```

:::

实测输出：

```
输入 16 条 → 送给模型 4 条
state 里仍是 17 条（原文保留）
存档文件：[PosixPath('.../conversation_history/session_7eea9013....md')]
```

> 🇹🇸 **TS 侧这组数字不一样，别以为是 bug**：TS 的假模型抓到 **2 次调用**（第 0 次是摘要模型那次，第 1 次才是主调用，7 条）。原因有两个：
> - **TS 的 `messages` 不含 system**。Python 的 `SENT[0]` 里有 1 条 `system` + 2 条保留 + 1 条新消息 = 4；TS 同样内容只算 3 条消息，加上 TS 侧压缩后还会插一条"你正在一段被压缩过的对话中"的引导语，主调用就是 7 条（`keep=6`）。
> - **`SENT[0]` 在 TS 里指向摘要模型的调用**。所以课文里 Python 写 `SENT[0]`，TS 必须写 `SENT[SENT.length - 1]`。
>
> 实测对照（同一份 16 条输入）：
>
> ```text [TypeScript]
> keep=2 → Spy 被调用 2 次 / state 17 条
>   [0] 1 条 ｜ 首条 human "You are a conversation summarizer. Your task is to"   ← 摘要模型
>   [1] 3 条 ｜ 首条 human "You are in the middle of a conversation that has b"   ← 主调用
> keep=6 → 主调用 7 条
> ```

> 🎯 **会用标志**：看到"输入 16 条、送模型 4 条、state 仍 17 条"这组数字，你能立刻说出：压缩发生在渲染层，不动存储层——所以原始日志还能用来做评测。

**变化二：验证工具结果卸载的边界**

::: code-group

```typescript [TypeScript]
import { z } from "zod";

const hugeTool = tool(
  async ({ n }) => "X".repeat(n),
  {
    name: "huge_tool",

    schema: z.object({ n: z.number() }),
  },
);
```

:::

实测三种情况：

```
自定义工具返回 400,000 字符
  → 交给模型 1,588 字符（卸载到 /large_tool_results/call_1）

read_file 读 600,015 字节文件
  → 交给模型 455 字符（未卸载，靠 offset/limit 分页）
  → /large_tool_results 目录根本不存在

TOOLS_EXCLUDED_FROM_EVICTION
  → ('ls', 'glob', 'grep', 'read_file', 'edit_file', 'write_file', 'delete')
```

> 🇹🇸 **TS 侧这条规则更严：只认 `ToolMessage`**。同一个 400,000 字符的结果，如果工具直接 `return "X".repeat(400000)`（裸字符串），TS 会**原样交给模型**，一点都不卸载：
>
> ```text [TypeScript]
> A) 裸字符串 400,000 字符 → 原样返回 400,000 字符（不卸载）
> B) ToolMessage 400,000 字符 → 交给模型 1,610 字符
>    large_tool_results 目录被创建: true
>    落盘文件: call_2.txt
> ```
>
> 实际用 `tool()` 包装时框架会自动包成 `ToolMessage`，所以正常写法不会踩到；但如果你手写中间件、在 `wrapToolCall` 里直接返回字符串，卸载就静默失效了。排除名单也略有不同：TS 是 `FILESYSTEM_TOOL_NAMES` 减去 `execute`。

> 🔍 **这条改变了工具设计思路**：之前你可能觉得"工具可以随便返回大结果，反正会卸载"。实际上，**自己写的工具最好支持分页**——卸载是被动补救，分页是主动控制。用户看到的是 455 字符而不是 1,588 字符的预览，差别很大。

**变化三：验证缓存断点，并避开自伤陷阱**

::: code-group

```python [Python]
from langchain_anthropic import ChatAnthropic


class SpyAnthropic(ChatAnthropic):
    """真实 ChatAnthropic 子类，但不发网络请求。"""

    @property
    def _llm_type(self):
        return "spy-anthropic"

    def _generate(self, messages, stop=None, run_manager=None, **kwargs):
        SENT.append(list(messages))
        return ChatResult(generations=[ChatGeneration(message=AIMessage(content="ok"))])


def check_cache_control(model, thread="C1"):
    ag = create_deep_agent(model=model, backend=fsb, memory=["/AGENTS.md"])
    SENT.clear()
    ag.invoke({"messages": [{"role": "user", "content": "go"}]},
              config={"configurable": {"thread_id": thread}})
    s = next((m for m in SENT[0] if m.type == "system"), None)
    blocks = s.content if isinstance(s.content, list) else [{"text": str(s.content)}]
    return [b.get("cache_control") for b in blocks if isinstance(b, dict)]


# Anthropic：应该有断点
print(check_cache_control(SpyAnthropic(model="claude-sonnet-4-5-20250929", api_key="sk-ant-fake")))
# → [{'type': 'ephemeral'}]

# 非 Anthropic：应该没有
print(check_cache_control(Spy()))
# → [None]
```

```typescript [TypeScript]
import { ChatAnthropic } from "@langchain/anthropic";

class SpyAnthropic extends ChatAnthropic {
  constructor(fields: ConstructorParameters<typeof ChatAnthropic>[0]) {
    super(fields);
  }
  _llmType() { return "spy-anthropic"; }
  async _generate(messages: BaseMessage[]): Promise<ChatResult> {
    SENT.push([...messages]);
    return { generations: [{ text: "ok", message: new AIMessage("ok") }] } as ChatResult;
  }
}

const checkCacheControl = async (model: BaseChatModel, thread = "C1"): Promise<unknown[]> => {
  const ag = await createDeepAgent({ model: model as never, backend: fsb, memory: ["/AGENTS.md"] });
  SENT.length = 0;
  await ag.invoke({ messages: [{ role: "user", content: "go" }] }, { configurable: { thread_id: thread } });
  const s = SENT[0].find((m) => m.getType() === "system");
  const c = s?.content;
  const blocks = Array.isArray(c) ? c : [{ text: String(c ?? "") }];
  return blocks.map((b) => (b as { cache_control?: unknown }).cache_control);
};

// Anthropic：应该有断点
console.log(await checkCacheControl(
  new SpyAnthropic({ model: "claude-sonnet-4-5-20250929", apiKey: "sk-ant-fake" }),
));
// → [{ type: 'ephemeral' }]

// 非 Anthropic：应该没有
console.log(await checkCacheControl(new Spy() as never));
// → [undefined]
```

:::

> 🇹🇸 **TS 侧的缓存中间件按 provider 装配，名单和 Python 不同**。实测四条栈：
>
> ```text [TypeScript]
> Anthropic  栈 7 个 | 缓存相关: PromptCachingMiddleware, CacheBreakpointMiddleware
> OpenAI     栈 5 个 | 缓存相关: （无）
> Bedrock    栈 6 个 | 缓存相关: BedrockPromptCachingMiddleware
> DeepSeek   栈 5 个 | 缓存相关: （无）
> ```
>
> Python 侧只有 `AnthropicPromptCachingMiddleware` 一个，挂在所有模型上（用 `ignore` 跳过非 Anthropic）；TS 是"哪个 provider 就装哪个"，OpenAI / DeepSeek 完全不装。**换成非 Anthropic 模型时，别指望 TS 会自动加断点。**
