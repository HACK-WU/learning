## 场景：5 个竞品同时查，查到一半要喊停

**场景**：你要调研 5 个竞品，每个都需要独立分析。而且这次调研本身是探索性的——很可能查到第 3 个就发现方向错了，需要改指令或喊停。

- **如果逐个查**：5 个竞品排队，中间 12 分钟在等，而且第 5 个已经错方向了还在跑
- **如果能同时查 + 能干预**：5 个一起提交，查的过程中能改、能停

**全貌一句话**：完整方案还包括把子代理部署成远程服务、把任务状态接到前端面板、给 `waiting` 状态配人工审批入口。本课先解决"怎么并发派、怎么量出收益、异步生命周期怎么走"。

### ① 基础实现（能跑但不知道它干了什么）

![基础版：一个循环派 5 个，主上下文只有一行](./assets/app-step1-dyn-fanout.svg)

> 看图：左边是一段 JS 循环派 5 个子代理，右边是主代理实际收到的内容——不是 5 份报告，而是一行统计。这个"看起来什么都没拿到"的现象，正是本课最核心的机制。

::: code-group

```python [Python]
import asyncio
import tempfile
from pathlib import Path

from langchain_core.language_models.chat_models import BaseChatModel
from langchain_core.messages import AIMessage, HumanMessage
from langchain_core.outputs import ChatGeneration, ChatResult
from langgraph.checkpoint.memory import InMemorySaver

from deepagents import create_deep_agent
from deepagents.backends import FilesystemBackend
from langchain_quickjs import CodeInterpreterMiddleware

SEEN = []


class Dispatcher(BaseChatModel):
    """写一段循环派 5 个。"""

    @property
    def _llm_type(self):
        return "dispatcher"

    def bind_tools(self, tools, **kwargs):
        return self

    def _generate(self, messages, stop=None, run_manager=None, **kwargs):
        if not any(m.type == "tool" for m in messages):
            code = """
const competitors = ["飞书", "Notion", "Coda", "Airtable", "ClickUp"];
const reports = await Promise.all(
  competitors.map(name =>
    task({ description: `调研竞品 ${name} 的定价策略`, subagentType: "researcher" })
  )
);
`共 ${reports.length} 份报告，总长度 ${reports.join("").length} 字`;
"""
            return ChatResult(generations=[ChatGeneration(message=AIMessage(
                content="", tool_calls=[{"name": "eval", "args": {"code": code}, "id": "e1"}]))])
        return ChatResult(generations=[ChatGeneration(message=AIMessage(content="调研完成"))])


class Researcher(BaseChatModel):
    @property
    def _llm_type(self):
        return "researcher"

    def bind_tools(self, tools, **kwargs):
        return self

    def _generate(self, messages, stop=None, run_manager=None, **kwargs):
        d = next((str(m.content) for m in messages if m.type == "human"), "?")
        SEEN.append(d)
        return ChatResult(generations=[ChatGeneration(message=AIMessage(
            content=f"【{d[:12]}】定价详情：" + "分层定价与席位计费差异明显，" * 12))])


root = Path(tempfile.mkdtemp())
agent = create_deep_agent(
    model=Dispatcher(),
    backend=FilesystemBackend(root_dir=str(root), virtual_mode=True),
    subagents=[{"name": "researcher", "description": "调研单个竞品", "model": Researcher()}],
    middleware=[CodeInterpreterMiddleware()],
    checkpointer=InMemorySaver(),
)

out = asyncio.run(agent.ainvoke(
    {"messages": [HumanMessage(content="调研这 5 个竞品")]},
    config={"configurable": {"thread_id": "comp-1"}}))

print("子代理被派活:", len(SEEN))
for s in SEEN:
    print("  -", s)
print("\n主上下文消息数:", len(out["messages"]))
print("主上下文总字符:", sum(len(str(m.content)) for m in out["messages"]))
print("最后消息:", str(out["messages"][-1].content)[:120])
```

```typescript [TypeScript]
import { BaseChatModel } from "@langchain/core/language_models/chat_models";
import { AIMessage, HumanMessage, type BaseMessage } from "@langchain/core/messages";
import type { ChatResult } from "@langchain/core/outputs";
import { MemorySaver } from "@langchain/langgraph";
import { createCodeInterpreterMiddleware } from "@langchain/quickjs";
import { mkdtempSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { createDeepAgent, FilesystemBackend } from "deepagents";

const SEEN: string[] = [];

// 这段 JS 是两侧通用的：subagentType 是驼峰，Python 侧也认
const code = `
const competitors = ["飞书", "Notion", "Coda", "Airtable", "ClickUp"];
const reports = await Promise.all(
  competitors.map(name =>
    task({ description: \`调研竞品 \${name} 的定价策略\`, subagentType: "researcher" })
  )
);
\`共 \${reports.length} 份报告，总长度 \${reports.join("").length} 字\`;
`;

class Dispatcher extends BaseChatModel {
  constructor() { super({}); }
  _llmType() { return "dispatcher"; }
  bindTools() { return this; }
  async _generate(messages: BaseMessage[]): Promise<ChatResult> {
    const hasTool = messages.some((m) => m.getType() === "tool");
    const msg = hasTool
      ? new AIMessage("调研完成")
      : new AIMessage({ content: "", tool_calls: [{ name: "eval", args: { code }, id: "e1" }] });
    return { generations: [{ text: String(msg.content), message: msg }] } as ChatResult;
  }
}

class Researcher extends BaseChatModel {
  constructor() { super({}); }
  _llmType() { return "researcher"; }
  bindTools() { return this; }
  async _generate(messages: BaseMessage[]): Promise<ChatResult> {
    const d = String(messages.find((m) => m.getType() === "human")?.content ?? "?");
    SEEN.push(d);
    const text = `【${d.slice(0, 12)}】定价详情：` + "分层定价与席位计费差异明显，".repeat(12);
    return { generations: [{ text, message: new AIMessage(text) }] } as ChatResult;
  }
}

const root = mkdtempSync(join(tmpdir(), "dyn-"));
const agent = await createDeepAgent({
  model: new Dispatcher() as never,
  backend: new FilesystemBackend({ rootDir: root, virtualMode: true }),
  subagents: [{ name: "researcher", description: "调研单个竞品", model: new Researcher() as never }],
  middleware: [createCodeInterpreterMiddleware() as never],
  checkpointer: new MemorySaver(),
});

const out = await agent.invoke(
  { messages: [new HumanMessage("调研这 5 个竞品")] },
  { configurable: { thread_id: "comp-1" } },
);

console.log("子代理被派活:", SEEN.length);
SEEN.forEach((s) => console.log("  -", s));
const msgs = out.messages as BaseMessage[];
console.log("\n主上下文消息数:", msgs.length);
console.log("主上下文总字符:", msgs.reduce((n, m) => n + String(m.content).length, 0));
console.log("最后消息:", String(msgs[msgs.length - 1].content).slice(0, 120));
```

:::

实测输出（两侧一致，TS 侧主上下文 4 条 / 32 字符）：

```
子代理被派活: 5
  - 调研竞品 飞书 的定价策略
  - 调研竞品 Notion 的定价策略
  ...
主上下文消息数: 4
主上下文总字符: 32
最后消息: 调研完成
```

> 🇹🇸 TS 侧两个 API 名字要换：中间件是 `createCodeInterpreterMiddleware()`（来自 `@langchain/quickjs`），`import` 路径也是 `@langchain/quickjs` 而不是 `langchain_quickjs`。**但里面那段 JS 代码两侧完全通用**——`task({ description, subagentType })` 在 Python 的 JS 解释器里同样有效。

> ⚠️ **它的三个问题**：
> 1. **看不出省了什么**。主上下文只有一行，但子代理明明返回了 5 份长报告——需要对照才能理解。
> 2. **不知道它比并行 `task` 慢**。这段代码在 5 个子代理的场景下，比直接发 5 个 `task` 调用更慢。
> 3. **异步那条路完全没走**。动态派发是"同一进程内并发"，没有 task_id，不能中途改指令、不能取消。

### ② 综合实现（被问题逼出来的下一步）

![综合版：动态 vs 并行 task 的量化对比 + 异步五工具生命周期](./assets/app-step2-modes-compare.svg)

> 看图：上半是动态与并行 `task` 的双栏对比（耗时、上下文、决策次数三项量化），下半是异步任务的状态流转：`running → waiting → 人回答 → running → success`，以及 `update` 会中断旧 run 开新 run。

差距的来源一句话：**动态模式下，8 份子代理结果留在解释器的 JS 变量里**；并行模式下，8 份全都进了主代理的 `messages`。

但耗时反过来（N=4）：

```
并行 task tool_calls  0.087s
动态 eval + 循环      0.329s     ← 慢 3.8 倍
```

这是解释器启动 + 序列化往返的固定开销，不随 N 摊薄。

> 🇹🇸 **TS 侧把这个"固定开销"量细了，结论比 Python 更乐观**：小 N 时动态略慢，但 N 一上来就反超——
>
> ```text [TypeScript]
> N=4   并行 0.038s  vs  动态 0.042s  → 动态慢 1.11x（3 次中位数）
> N=12  并行 0.052s  vs  动态 0.049s  → 动态快 1.06x
> N=30  并行 0.088s  vs  动态 0.080s  → 动态快 1.10x
> ```
>
> 原因是 TS 的 QuickJS 是**长期驻留**的（不像 Python 侧每次 `eval` 都新起解释器），所以固定开销只付一次。**选型结论因此变了**：TS 侧只要 N ≥ 8 就可以放心用动态模式，不必为了性能退回并行 `task`。

**变化二：把 `waiting` 状态搞清楚**

异步子代理自己也能配审批（课 11 的知识）。它暂停时，run 状态是 `success`，只有 thread 显示它在等：

::: code-group

```python [Python]
_WAITING = "waiting"
_THREAD_TO_TASK_STATUS = {"interrupted": _WAITING, "busy": "running", "error": "error"}


def _task_status(run, thread):
    if run["status"] != "success" or thread is None:
        return run["status"]
    return _THREAD_TO_TASK_STATUS.get(thread.get("status"), "success")
```

```typescript [TypeScript]
const WAITING = "waiting";
const THREAD_TO_TASK_STATUS: Record<string, string> = {
  interrupted: WAITING, busy: "running", error: "error",
};

const taskStatus = (
  run: { status: string },
  thread: { status?: string } | null,
): string => {
  if (run.status !== "success" || thread === null) return run.status;
  return THREAD_TO_TASK_STATUS[thread.status ?? ""] ?? "success";
};
```

:::

`waiting` 的任务，主代理**不能代批**：

::: code-group

```python [Python]
_WAITING_NOTE = (
    "The subagent is waiting for a human. The user answers it directly; "
    "you can't answer or approve it."
)
```

```typescript [TypeScript]
const WAITING_NOTE =
  "The subagent is waiting for a human. The user answers it directly; " +
  "you can't answer or approve it.";
```

:::

> 🇹🇸 **TS 侧判定"这是不是异步子代理"的开关是 `graphId`，不是 `url`**，而且判据只是 `"graphId" in spec`（不做非空校验）：
>
> ```text [TypeScript]
> {name, graphId}   isAsyncSubAgent = true
> {name, url}       isAsyncSubAgent = false      ← 只写 url 不算
> {name}            isAsyncSubAgent = false
> {"graphId":""}    isAsyncSubAgent = true       ← 空串也算
> ```
>
> 配上异步子代理后，工具从 **9 个变 14 个**，新增的五个名字与 Python 一字不差：`start_async_task` / `check_async_task` / `update_async_task` / `cancel_async_task` / `list_async_tasks`。中间件栈也从 5 条变 6 条（`asyncSubAgentMiddleware` 插在 `patchToolCallsMiddleware` 之后）。

**变化三：`update` 和 `cancel` 的真实语义**

::: code-group

```python [Python]
# update：中断旧 run，同 thread 开新 run（task_id 不变）
run = client.runs.create(
    thread_id=tracked["thread_id"],
    assistant_id=spec["graph_id"],
    input={"messages": [{"role": "user", "content": message}]},
    multitask_strategy="interrupt",     # ← 关键
)

# cancel：如果任务在 waiting，不调远程 cancel（没有活跃 run）
if tracked["status"] != _WAITING:
    client.runs.cancel(thread_id=..., run_id=...)
```

```typescript [TypeScript]
// update：中断旧 run，同 thread 开新 run（task_id 不变）
const run = await client.runs.create(tracked.threadId, tracked.graphId, {
  input: { messages: [{ role: "user", content: message }] },
  multitaskStrategy: "interrupt",      // ← 关键，驼峰
});

// cancel：如果任务在 waiting，不调远程 cancel（没有活跃 run）
if (tracked.status !== WAITING) {
  await client.runs.cancel(tracked.threadId, tracked.runId);
}
```

:::

> 🇹🇸 TS 侧异步五工具的实测语义（假 Agent Protocol 服务驱动，全离线）：
>
> ```text [TypeScript]
> start   → "Launched async subagent. taskId: t-1"｜taskId 就是 threadId
> check   → {"status":"pending","threadId":"t-1"}   （pending 时没有 result 字段）
>           成功后 {"status":"success", ..., "result":"远程图 research_agent 完成..."}
>           状态会回写进 asyncTasks
> update  → taskId 不变，**runId 变了**（r-1 → r-2），远程收到 multitask=interrupt
> list    → 按"缓存状态"筛，不是按实时状态筛（缓存说 pending，远程已 success 也显示 pending）
> cancel  → "Cancelled async subagent task: t-1"｜远程 cancel 调用 1 次
> ```
>
> `list` 那条要特别注意：**它读的是本地缓存的 `asyncTasks`，不是去远程查**。所以想拿实时状态必须自己调 `check`。

**再踩一个坑**：不配 `url` 用同步 `invoke`，报错被吞成工具返回值，流程不中断：

```
ToolMessage: Failed to launch async subagent 'r': Async subagent 'r' has no url
configured. ASGI transport (url=None) requires async invocation.
```

> 🇹🇸 **TS 侧连这条提示都没有**——`url` 缺失时 SDK 直接回落到 `http://localhost:8123`，报的是连接失败：
>
> ```text [TypeScript]
> tool = Failed to launch async subagent 'r': ConnectionError: Unable to connect to
>        LangGraph server. Please ensure the server is running and accessible.
> 有没有 "requires async invocation" 提示: false
> ```
>
> 🇹🇸 所以 TS 侧看到 `ConnectionError: Unable to connect to LangGraph server` 时，**第一反应应该是"我是不是忘了配 url"**，而不是去查服务有没有起。另外两种错误形态也不同：未知 agentName 与未知 taskId 在 TS 侧都是普通字符串返回（`Unknown async subagent type \`nobody\`. Available types: ...`、`No tracked task found for taskId: 't-999'`），流程不中断、不抛错。

**补充：动态 `task()` 的四个实测细节（TS 侧）**

::: code-group

```python [Python]
# Python 侧解释器全局里能拿到的东西（JS 侧同名）
# __sessionId__ / console / performance / task
# 没有 require / process / fetch / Deno —— 是隔离沙箱
```

```typescript [TypeScript]
// 解释器全局里能拿到的东西
console.log(Object.keys(globalThis).sort().join(","));
// → __sessionId__,console,performance,task
// 边界检查：
[typeof require, typeof process, typeof fetch, typeof Deno].join(" | ");
// → undefined | undefined | undefined | undefined
```

:::

四条结论：

- **并发上限默认 8**。实测 8 个子代理每个睡 60ms：默认并发峰值 8、耗时 0.10s；设 `maxSubagentConcurrency: 2` 后峰值 2、耗时 0.28s。
- **子代理抛异常会穿到 JS 侧**（`抛到 JS 侧了: worker 内部炸了`），所以 `task()` 必须包 `try/catch`。
- **键名 `subagentType` 和 `subagent_type` 都收**，但报错提示只列驼峰：`task: unknown keys: bogus. Allowed: description, subagentType, responseSchema`。
- **`responseSchema` 能拿结构化对象**（`typeof = object`），不是字符串——这是让子代理返回可解析结果的唯一正路。
