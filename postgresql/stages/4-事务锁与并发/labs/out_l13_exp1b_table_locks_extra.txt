实验 A · 各命令在 finance.accounts 上自动获取的表级锁
==============================================================================
  （BEGIN → 执行命令 → 查 pg_locks → ROLLBACK）

  SELECT                         → AccessShareLock
  UPDATE                         → RowExclusiveLock
  DELETE                         → RowExclusiveLock
  INSERT                         → RowExclusiveLock
  SELECT FOR UPDATE              → RowShareLock
  SELECT FOR NO KEY UPDATE       → RowShareLock
  SELECT FOR SHARE               → RowShareLock
  SELECT FOR KEY SHARE           → RowShareLock
  CREATE INDEX                   → ShareLock
  ALTER TABLE ADD COLUMN         → AccessExclusiveLock
  TRUNCATE                       → AccessExclusiveLock + ShareLock
  ANALYZE                        → ShareUpdateExclusiveLock
  LOCK TABLE（不写模式）               → AccessExclusiveLock
  CREATE TRIGGER 类（SHARE ROW EXCL） → ShareRowExclusiveLock

  --- 显式 LOCK TABLE 八种模式 ---
  LOCK TABLE ... IN ACCESS SHARE MODE        → AccessShareLock
  LOCK TABLE ... IN ROW SHARE MODE           → RowShareLock
  LOCK TABLE ... IN ROW EXCLUSIVE MODE       → RowExclusiveLock
  LOCK TABLE ... IN SHARE UPDATE EXCLUSIVE MODE → ShareUpdateExclusiveLock
  LOCK TABLE ... IN SHARE MODE               → ShareLock
  LOCK TABLE ... IN SHARE ROW EXCLUSIVE MODE → ShareRowExclusiveLock
  LOCK TABLE ... IN EXCLUSIVE MODE           → ExclusiveLock
  LOCK TABLE ... IN ACCESS EXCLUSIVE MODE    → AccessExclusiveLock
